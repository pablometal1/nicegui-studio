from nicegui import ui
from forms.clientes_designer import ClientesFormDesigner


class ClientesForm(ClientesFormDesigner):
    def __init__(self) -> None:
        super().__init__()
        self.setup_ui()

    def on_btn_guardar_click(self, e) -> None:
        nombre = self.input_nombre.value
        if not nombre:
            ui.notify("Ingresá un nombre", type="warning")
            return
        ui.notify(f"Cliente {nombre} guardado!", type="positive")

from nicegui import ui


class ClientesFormDesigner:
    def setup_ui(self) -> None:
        ui.query("body").style("background: #0f172a;")
        self.root = ui.column().classes("w-full max-w-4xl mx-auto p-6 gap-4")
        with self.root:
            with ui.card().classes("w-full p-6 gap-4"):
                self.input_nombre = ui.input("Nombre").props("outlined dense")
                self.btn_guardar = ui.button(
                    "Guardar",
                    color="#2563eb",
                    on_click=self.on_btn_guardar_click,
                ).props("unelevated no-caps")

PREGUNTAS FRECUENTES

Todo lo que necesitás saber

¿Puedo usar Quantum-RAD Studio para proyectos comerciales?

Sí. Tanto Quantum-RAD Studio como NiceGUI utilizan licencia MIT, lo que te permite construir, empaquetar y vender tus aplicaciones libremente.

¿Qué pasa si edito manualmente el código Python de mi formulario?

Tu lógica vive en forms/<nombre>.py y nunca es sobrescrita por el diseñador. Solo se regenera forms/<nombre>_designer.py.

¿Las aplicaciones creadas dependen de Quantum-RAD Studio para ejecutarse?

No. Cada proyecto es una aplicación Python + NiceGUI 100% independiente que podés ejecutar con python main.py, en Docker o compilada como .exe.

¿Está pensado para uso local o en un servidor público?

Quantum-RAD Studio es un IDE de desarrollo local (127.0.0.1). Las aplicaciones que construís con él sí pueden desplegarse en cualquier servidor web o escritorio.
